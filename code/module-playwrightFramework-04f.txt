async function dismissDemoQAAds(page: Page): Promise<void> {
  // If page already closed, do nothing
  if (page.isClosed()) return;

  // 1) If vignette hash appears, try escape (no waiting)
  try {
    if (!page.isClosed() && page.url().includes("google_vignette")) {
      await page.keyboard.press("Escape");
    }
  } catch {}

  // If page got replaced/closed, stop
  if (page.isClosed()) return;

  // 2) Close fixed bottom banner if present (no waitForTimeout)
  try {
    const closeBtn = page.locator("#close-fixedban");
    if (!page.isClosed() && (await closeBtn.count()) > 0) {
      // Use first() to avoid strict-mode issues
      await closeBtn
        .first()
        .click({ timeout: 1000 })
        .catch(() => {});
    }
  } catch {}

  if (page.isClosed()) return;

  // 3) Hide common overlay containers (best-effort)
  try {
    await page.evaluate(() => {
      const selectors = [
        "#ad_position_box",
        'div[id^="google_ads_iframe_"]',
        'div[id$="__container__"]',
        'iframe[id^="google_ads_iframe_"]',
        'iframe[src*="googlesyndication"]',
      ];

      for (const sel of selectors) {
        document.querySelectorAll(sel).forEach((el) => {
          const node = el as HTMLElement;
          node.style.setProperty("display", "none", "important");
          node.style.setProperty("visibility", "hidden", "important");
          node.style.setProperty("pointer-events", "none", "important");
          node.style.setProperty("opacity", "0", "important");
        });
      }

      // restore scroll/click
      document.body.style.overflow = "auto";
      document.documentElement.style.overflow = "auto";
    });
  } catch {}
}

test("TC01 - Prerequisite Website Navigation to Tabs Page", async ({
  page,
}, testInfo) => {
  testMetadata04f(testInfo);

  await page.goto("https://demoqa.com/tabs", {
    waitUntil: "domcontentloaded",
  });

  // dismiss Google ads if appear
  await dismissDemoQAAds(page);

  // If ads re-appear right before click, dismiss again
  await dismissDemoQAAds(page);

  // 1️⃣ Page exists
  expect(page).toBeTruthy();

  // 2️⃣ Correct page loaded (auto-waits)
  await expect(page).toHaveURL(/\/tabs/);

  // 3️⃣ Critical element exists (more stable)
  const heading = page.locator("h1.text-center");
  await expect(heading).toHaveText("Tabs");
});

async function getFirstTabValue(page: Page): Promise<string> {
  const whatTab = page.locator("#demo-tab-what");
  const whatPanel = page.locator("#demo-tabpane-what");

  await whatTab.scrollIntoViewIfNeeded();

  // quick ad dismissal just before click (optional but helps stability)
  await dismissDemoQAAds(page);

  await whatTab.click({ timeout: 5000 }).catch(async () => {
    await dismissDemoQAAds(page);
    await whatTab.click({ timeout: 5000, force: true });
  });

  // Wait until panel is active (more reliable than visibility)
  await expect(whatPanel).toHaveAttribute("aria-hidden", "false", {
    timeout: 8000,
  });

  // Faster than innerText()
  const text = await whatPanel.textContent();
  return (text ?? "").trim();
}

async function getSecondTabValue(page: Page): Promise<string> {
  const originTab = page.locator("#demo-tab-origin");
  const originTabPanel = page.locator("#demo-tabpane-origin");

  // Scroll into view
  await originTab.scrollIntoViewIfNeeded();

  // Always try to dismiss ads first
  await dismissDemoQAAds(page);

  // --- Attempt 1: normal click
  await originTab.click({ timeout: 5000 }).catch(() => {});

  // If not activated, attempt 2
  const selected1 = await originTab.getAttribute("aria-selected");
  if (selected1 !== "true") {
    await dismissDemoQAAds(page);

    // --- Attempt 2: force click (last resort for ad overlays / actionability issues)
    await originTab.click({ timeout: 5000, force: true }).catch(() => {});
  }

  // Wait until the panel is activated (more reliable than waiting aria-selected)
  await expect(originTabPanel).toHaveAttribute("aria-hidden", "false", {
    timeout: 10000,
  });

  // Extract both paragraphs inside Origin tab
  const paragraphs = originTabPanel.locator("p");

  const text1 = (await paragraphs.nth(0).innerText()).trim();
  const text2 = (await paragraphs.nth(1).innerText()).trim();

  return `${text1}\n${text2}`;
}

  const useTab = page.locator("#demo-tab-use");
  const useTabPanel = page.locator("#demo-tabpane-use");

  // Scroll into view to avoid interception
  await useTab.scrollIntoViewIfNeeded();

  // Dismiss ads if they appear (your helper)
  await dismissDemoQAAds(page);

  // Try normal click, fallback to force click if needed
  try {
    await useTab.click({ timeout: 5000 });
  } catch {
    await dismissDemoQAAds(page);
    await useTab.click({ timeout: 5000, force: true });
  }

  // Wait until Use tab panel is activated (more reliable than toBeVisible)
  await expect(useTabPanel).toHaveAttribute("aria-hidden", "false", {
    timeout: 10000,
  });

  // Extract paragraph text (p.mt-3)
  const paragraph = useTabPanel.locator("p.mt-3");

  // textContent() is faster than innerText()
  const text = await paragraph.textContent();
  return (text ?? "").trim();
}

async function getFourthTabValue(page: Page): Promise<string> {
  try {
    const moreTabPanel = page.locator("#demo-tabpane-more");

    // Check if panel exists in DOM
    const count = await moreTabPanel.count();
    if (count === 0) {
      return "";
    }

    // Check visibility (disabled tab usually keeps panel hidden)
    if (await moreTabPanel.isVisible()) {
      const text = await moreTabPanel.textContent();
      return (text ?? "").trim();
    }
  } catch {
    // Expected behavior:
    // - tab not clickable
    // - panel not visible
    // - element not interactable
  }

  // Disabled tab → return blank
  return "";
}

test("TC05 - Verify Fourth Tab (Disabled) – Negative Scenario", async ({
  page,
}) => {
  await page.goto("https://demoqa.com/tabs", { waitUntil: "domcontentloaded" });

  const text = await getFourthTabValue(page);

  // Negative scenario requirement:
  // If tab is disabled -> text returns "" -> FAIL (expected negative)
  expect(text, "More tab is not clickable as expected").not.toBe("");
});
