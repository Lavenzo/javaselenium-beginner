// Test File
import { test, expect, type Page } from "@playwright/test";
import { testMetadata03c } from "./utils/testMetadata03a";

test.describe("TC01 - Prerequisite Website Navigation", () => {
  test("verifyPrereqIsWorking", async ({ page }, testInfo) => {
    testMetadata03c(testInfo);

    await page.goto("https://demoqa.com/modal-dialogs", {
      waitUntil: "domcontentloaded",
    });

    // 1️⃣ Page exists
    expect(page).toBeTruthy();

    // 2️⃣ Correct page loaded (auto-waits)
    await expect(page).toHaveURL(/\/modal-dialogs/);

    // 3️⃣ Critical element exists
    const mainContent = page.locator("#modalWrapper h1", {
      hasText: "Modal Dialogs",
    });
    await expect(mainContent).toBeVisible();
  });
});

async function getExternalModalText(page: Page): Promise<string> {
  // 1️⃣ Ensure we are in main DOM
  // (Playwright is already in main DOM unless frameLocator is used)

  // 2️⃣ Locate text inside Modal Dialogs section
  const modalText = page.locator(
    "xpath=//div[@id='modalWrapper']//div[normalize-space()='Click on button to see modal']"
  );

  // 3️⃣ Wait until visible
  await expect(modalText).toBeVisible();

  // 4️⃣ Return trimmed text
  return (await modalText.textContent())!.trim();
}

test("TC02 - Verify External Modal Text", async ({ page }, testInfo) => {
  testMetadata03c(testInfo);
  await page.goto("https://demoqa.com/modal-dialogs", {
    waitUntil: "domcontentloaded",
  });

  const externalText = await getExternalModalText(page);

  expect(externalText).toBe("Click on button to see modal");
});

async function getSmallModalText(page: Page): Promise<string> {
  // 1️⃣ Ensure we are on main DOM
  // (Playwright is already on main DOM unless you use frameLocator)

  // 2️⃣ Click Small Modal button
  await page.locator("#showSmallModal").click();

  // 3️⃣ Wait for modal body text to appear and capture it
  const modalBodyText = page.locator("div.modal-body");
  await expect(modalBodyText).toBeVisible();

  const text = (await modalBodyText.textContent())!.trim();

  // 4️⃣ Hard wait 500ms before closing (as requested)
  await page.waitForTimeout(500);

  // 5️⃣ Click Close button
  await page.locator("#closeSmallModal").click();

  // 6️⃣ Ensure modal is closed
  await expect(page.locator("div.modal-dialog")).toBeHidden();

  return text;
}

test("TC03 - Verify Small Modal Text", async ({ page }, testInfo) => {
  testMetadata03c(testInfo);
  await page.goto("https://demoqa.com/modal-dialogs", {
    waitUntil: "domcontentloaded",
  });

  const childText = await getSmallModalText(page);

  expect(childText).toBe("This is a small modal. It has very less content");
});
