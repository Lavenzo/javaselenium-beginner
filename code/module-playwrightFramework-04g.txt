async function dismissDemoQAAds(page: Page): Promise<void> {
  // If page already closed, do nothing
  if (page.isClosed()) return;

  // 1) If vignette hash appears, try escape (no waiting)
  try {
    if (!page.isClosed() && page.url().includes("google_vignette")) {
      await page.keyboard.press("Escape");
    }
  } catch {}

  // If page got replaced/closed, stop
  if (page.isClosed()) return;

  // 2) Close fixed bottom banner if present (no waitForTimeout)
  try {
    const closeBtn = page.locator("#close-fixedban");
    if (!page.isClosed() && (await closeBtn.count()) > 0) {
      // Use first() to avoid strict-mode issues
      await closeBtn
        .first()
        .click({ timeout: 1000 })
        .catch(() => {});
    }
  } catch {}

  if (page.isClosed()) return;

  // 3) Hide common overlay containers (best-effort)
  try {
    await page.evaluate(() => {
      const selectors = [
        "#ad_position_box",
        'div[id^="google_ads_iframe_"]',
        'div[id$="__container__"]',
        'iframe[id^="google_ads_iframe_"]',
        'iframe[src*="googlesyndication"]',
      ];

      for (const sel of selectors) {
        document.querySelectorAll(sel).forEach((el) => {
          const node = el as HTMLElement;
          node.style.setProperty("display", "none", "important");
          node.style.setProperty("visibility", "hidden", "important");
          node.style.setProperty("pointer-events", "none", "important");
          node.style.setProperty("opacity", "0", "important");
        });
      }

      // restore scroll/click
      document.body.style.overflow = "auto";
      document.documentElement.style.overflow = "auto";
    });
  } catch {}
}

test("TC01 - Prerequisite Website Navigation to Tool Tips Page", async ({
  page,
}, testInfo) => {
  testMetadata04g(testInfo);

  await page.goto("https://demoqa.com/tool-tips", {
    waitUntil: "domcontentloaded",
  });

  // dismiss Google ads if appear
  await dismissDemoQAAds(page);

  // If ads re-appear right before click, dismiss again
  await dismissDemoQAAds(page);

  // 1️⃣ Page exists
  expect(page).toBeTruthy();

  // 2️⃣ Correct page loaded (auto-waits)
  await expect(page).toHaveURL(/\/tool-tips/);

  // 3️⃣ Critical element exists (more stable)
  const heading = page.locator("h1.text-center");
  await expect(heading).toHaveText("Tool Tips");
});

async function hoverAndGetTooltipText(
  page: Page,
  target: Locator,
  tooltipInner: Locator,
  timeoutMs = 15_000
): Promise<string> {
  await target.waitFor({ state: "visible", timeout: timeoutMs });

  await target.scrollIntoViewIfNeeded();

  // Let page settle (ads/reflow)
  await page.waitForLoadState("domcontentloaded");
  await page.waitForTimeout(200);

  // Optional but helps with intermittent overlay issues
  await waitForNotCovered(target, timeoutMs);

  // Try hover up to 3 times
  for (let i = 0; i < 3; i++) {
    try {
      await target.hover({ position: { x: 1, y: 1 }, timeout: timeoutMs });

      await tooltipInner.waitFor({ state: "visible", timeout: 2_000 });
      const txt = (await tooltipInner.innerText()).trim();
      if (txt) return txt;
    } catch {
      await page.waitForTimeout(200);
    }
  }

  // Last resort - force hover with JS events
  await forceHoverWithJS(target);

  await tooltipInner.waitFor({ state: "visible", timeout: timeoutMs });
  return (await tooltipInner.innerText()).trim();
}

async function waitForNotCovered(
  target: Locator,
  timeoutMs: number
): Promise<void> {
  const start = Date.now();

  while (Date.now() - start < timeoutMs) {
    const ok = await target.evaluate((el) => {
      const r = el.getBoundingClientRect();
      const x = r.left + Math.min(5, r.width / 2);
      const y = r.top + Math.min(5, r.height / 2);
      const topEl = document.elementFromPoint(x, y);
      return topEl === el || (topEl && el.contains(topEl));
    });

    if (ok) return;
    await new Promise((res) => setTimeout(res, 150));
  }
}

async function forceHoverWithJS(target: Locator): Promise<void> {
  await target.evaluate((el) => {
    const rect = el.getBoundingClientRect();
    const x = rect.left + Math.min(5, rect.width / 2);
    const y = rect.top + Math.min(5, rect.height / 2);

    const opts: MouseEventInit = {
      bubbles: true,
      cancelable: true,
      clientX: x,
      clientY: y,
    };
    el.dispatchEvent(new MouseEvent("mouseover", opts));
    el.dispatchEvent(new MouseEvent("mousemove", opts));
    el.dispatchEvent(new MouseEvent("mouseenter", opts));
  });
}

/** Specific helper for your button tooltip (matches your Java method getHoverButtonText) */
async function getHoverButtonText(page: Page): Promise<string> {
  return hoverAndGetTooltipText(
    page,
    page.locator("#toolTipButton"),
    page.locator("#buttonToolTip .tooltip-inner"),
    15_000
  );
}

    15_000
  );
}

  );
}
