/** Best-effort ads/overlay cleanup for DemoQA (safe to call any time) */
async function dismissDemoQAAds(page: Page): Promise<void> {
  if (page.isClosed()) return;

  // 1) If vignette hash appears, try escape
  try {
    if (!page.isClosed() && page.url().includes("google_vignette")) {
      await page.keyboard.press("Escape");
    }
  } catch {}

  if (page.isClosed()) return;

  // 2) Close fixed bottom banner if present
  try {
    const closeBtn = page.locator("#close-fixedban");
    if (!page.isClosed() && (await closeBtn.count()) > 0) {
      await closeBtn
        .first()
        .click({ timeout: 1000 })
        .catch(() => {});
    }
  } catch {}

  if (page.isClosed()) return;

  // 3) Hide common overlay containers (best-effort)
  try {
    await page.evaluate(() => {
      const selectors = [
        "#ad_position_box",
        'div[id^="google_ads_iframe_"]',
        'div[id$="__container__"]',
        'iframe[id^="google_ads_iframe_"]',
        'iframe[src*="googlesyndication"]',
      ];

      for (const sel of selectors) {
        document.querySelectorAll(sel).forEach((el) => {
          const node = el as HTMLElement;
          node.style.setProperty("display", "none", "important");
          node.style.setProperty("visibility", "hidden", "important");
          node.style.setProperty("pointer-events", "none", "important");
          node.style.setProperty("opacity", "0", "important");
        });
      }

      document.body.style.overflow = "auto";
      document.documentElement.style.overflow = "auto";
    });
  } catch {}
}

async function getSelectOneDropdown(
  page: Page,
  title: string
): Promise<string> {
  const selectOne = page.locator("#selectOne");
  await expect(selectOne).toBeVisible();
  await selectOne.scrollIntoViewIfNeeded();

  // click the actual control area
  const control = selectOne.locator("div[class*='control']").first();
  await control.click({ force: true });

  const input = selectOne.locator("input").first();
  await expect(input).toBeVisible({ timeout: 10000 });

  // force focus into the input (ads can intercept normal click)
  // force focus into the input (ads can intercept normal click)
  await input.click({ force: true });

  // clear + type + ENTER on the input itself (not page.keyboard)
  await input.fill(title);
  await input.press("Enter");

  const selectedValue = selectOne.locator("div[class*='singleValue']").first();
  await expect(selectedValue).toHaveText(title, { timeout: 10000 });
  return (await selectedValue.innerText()).trim();
}

  // ✅ Read visible text (Selenium equivalent)
  const selectedText = await dropdown.locator("option:checked").textContent();

  return selectedText?.trim() ?? "";
}

// ✅ Single implementation
export async function multiSelectDropdownSelection(
  page: Page,
  valueOne: string,
  valueTwo?: string,
  valueThree?: string,
  valueFour?: string
): Promise<string[]> {
  const colours = ["Green", "Blue", "Black", "Red"];

  const label = page.getByText(normalizeSpaceRegex("Multiselect drop down"));
  await expect(label).toBeVisible({ timeout: 15000 });

  const row = label.locator("xpath=ancestor::div[contains(@class,'row')][1]");
  await expect(row).toBeVisible({ timeout: 15000 });

  const control = row.locator("div[class*='control']").first();
  await expect(control).toBeVisible({ timeout: 15000 });
  await control.scrollIntoViewIfNeeded();
  await control.click({ force: true });

  const input = row.locator("input").first();
  await expect(input).toBeVisible({ timeout: 15000 });
  await input.click({ force: true });

  // Clear existing selections
  for (let i = 0; i < 10; i++) {
    const t = (await control.innerText()).trim();
    const hasAnyColour = colours.some((c) => t.includes(c));
    if (!hasAnyColour) break;
    await input.press("Backspace");
  }

  const pick = async (val: string) => {
    await input.fill(val);
    await input.press("Enter");
    await expect(control).toContainText(val, { timeout: 15000 });
  };

  // ✅ Validate no duplicates in provided values
  const requested = [valueOne, valueTwo, valueThree, valueFour].filter(
    (v): v is string => typeof v === "string" && v.trim().length > 0
  );

  const unique = new Set(requested);
  if (unique.size !== requested.length) {
    throw new Error(
      `multiSelectDropdownSelection: duplicate values detected: ${requested.join(
        ", "
      )}`
    );
  }

  // Select 1–4 values
  await pick(valueOne);
  if (valueTwo) await pick(valueTwo);
  if (valueThree) await pick(valueThree);
  if (valueFour) await pick(valueFour);

  // Read back selected values from control text
  const controlText = (await control.innerText()).trim();
  return colours.filter((c) => controlText.includes(c));
}

/** Best-effort ads/overlay cleanup for DemoQA (safe to call any time) */
async function dismissDemoQAAds(page: Page): Promise<void> {
  if (page.isClosed()) return;

  // 1) If vignette hash appears, try escape
  try {
    if (!page.isClosed() && page.url().includes("google_vignette")) {
      await page.keyboard.press("Escape");
    }
  } catch {}

  if (page.isClosed()) return;

  // 2) Close fixed bottom banner if present
  try {
    const closeBtn = page.locator("#close-fixedban");
    if (!page.isClosed() && (await closeBtn.count()) > 0) {
      await closeBtn
        .first()
        .click({ timeout: 1000 })
        .catch(() => {});
    }
  } catch {}

  if (page.isClosed()) return;

  // 3) Hide common overlay containers (best-effort)
  try {
    await page.evaluate(() => {
      const selectors = [
        "#ad_position_box",
        'div[id^="google_ads_iframe_"]',
        'div[id$="__container__"]',
        'iframe[id^="google_ads_iframe_"]',
        'iframe[src*="googlesyndication"]',
      ];

      for (const sel of selectors) {
        document.querySelectorAll(sel).forEach((el) => {
          const node = el as HTMLElement;
          node.style.setProperty("display", "none", "important");
          node.style.setProperty("visibility", "hidden", "important");
          node.style.setProperty("pointer-events", "none", "important");
          node.style.setProperty("opacity", "0", "important");
        });
      }

      document.body.style.overflow = "auto";
      document.documentElement.style.overflow = "auto";
    });
  } catch {}
}

export async function standardMultiSelect(
  page: Page,
  ...values: string[]
): Promise<string[]> {
  const select = page.locator("#cars");

  // 1) Ensure the element exists first
  await expect(select).toBeVisible({ timeout: 15000 });

  // 2) Scroll into view (Playwright built-in)
  await select.scrollIntoViewIfNeeded();

  // 3) Extra: center it in viewport (helps your screenshot/report and avoids ads covering it)
  await page.evaluate(() => {
    const el = document.querySelector("#cars");
    if (el) el.scrollIntoView({ block: "center", inline: "nearest" });
  });

  // (Optional but useful) small pause so the UI settles for screenshots/videos
  await page.waitForTimeout(300);

  // 4) Safety check - multiple attribute exists but value may be empty string -> just check it exists
  await expect(select).toHaveAttribute("multiple");

  // 5) Clear then select requested values by label
  await select.selectOption([]); // deselect all
  await select.selectOption(values.map((v) => ({ label: v })));

  // 6) Return selected option labels
  const selected = await select.evaluate((el) => {
    const s = el as HTMLSelectElement;
    return Array.from(s.selectedOptions).map((o) => o.text.trim());
  });

  return selected;
}
