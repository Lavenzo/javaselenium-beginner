// -------- Helper to fill Text Box form --------
async function textBoxInput(page: Page, iteration: number) {
  const randIndex = Math.floor(Math.random() * customerUserIDs.length);
  const fullName = customerUserIDs[randIndex];

  console.log(`\n[${iteration}/5] ✍ Filling Text Box: ${fullName}`);

  await page.evaluate(() => window.scrollBy(0, 250));
  await page.waitForTimeout(500);

  // Define the locators
  const name = page.locator("#userName");
  const email = page.locator("#userEmail");
  const currentAddr = page.locator("textarea#currentAddress");
  const permAddr = page.locator("textarea#permanentAddress");

  // Clear first (makes it obvious visually)
  await name.click();
  await page.keyboard.press("Control+A");
  await page.keyboard.press("Backspace");

  await email.click();
  await page.keyboard.press("Control+A");
  await page.keyboard.press("Backspace");

  await currentAddr.click();
  await page.keyboard.press("Control+A");
  await page.keyboard.press("Backspace");

  await permAddr.click();
  await page.keyboard.press("Control+A");
  await page.keyboard.press("Backspace");

  // Fill
  await name.type(fullName, { delay: 20 });
  await email.type(`${fullName}@gmail.com`, { delay: 20 });
  await currentAddr.type(`${fullName} Current Address Location.`, {
    delay: 10,
  });
  await permAddr.type(`${fullName} Permanent Address Location.`, { delay: 10 });

  // Submit
  const submitBtn = page.locator("button#submit");
  await submitBtn.scrollIntoViewIfNeeded();
  await submitBtn.click();

  // Wait until output box updates (proof that submit happened)
  await page.locator("#output").waitFor({ state: "visible" });
  const out = await page.locator("#output").innerText();
  console.log(`[${iteration}/5] ✅ Submitted. Output now:\n${out}`);
  await page.waitForTimeout(1000);

  // Scroll back up for next iteration
  await page.evaluate(() => window.scrollTo(0, 0));
}

// -------- Helper to close the Playwright script --------
async function waitForEnter(): Promise<void> {
  return new Promise((resolve) => {
    const rl = readline.createInterface({
      input: process.stdin,
      output: process.stdout,
    });
    rl.question("Browser is running. Press ENTER to close...\n", () => {
      rl.close();
      resolve();
    });
  });
}

async function openThirdTabThenClose(
  context: BrowserContext,
  childPage: Page, // tab 2 (current page)
  holdMs: number = 2000
) {
  console.log("Click simpleLink - open tab 3");

  const simpleLink = childPage.locator("#simpleLink").first();

  // Tab 3 is a "popup" from tab 2 (more accurate than context.waitForEvent("page"))
  const [thirdPage] = await Promise.all([
    childPage.waitForEvent("popup"),
    simpleLink.click(),
  ]);

  await thirdPage.waitForLoadState("domcontentloaded");
  await thirdPage.bringToFront();

  await thirdPage.waitForTimeout(holdMs);

  await thirdPage.close();

  // Back to tab 2
  await childPage.bringToFront();
  console.log("Back to tab 2:", childPage.url());
}

async function bypassAds(page: Page, linkLocator: string) {
  const link = page.locator(linkLocator).first();

  const href = await link.getAttribute("href");
  if (!href) {
    throw new Error(`No href found for locator: ${linkLocator}`);
  }

  // Go directly instead of clicking (bypass ad interstitial)
  await page.goto(href, { waitUntil: "domcontentloaded" });

  // Safety net: if Google vignette still hijacks
  if (page.url().includes("google_vignette")) {
    console.log("Google vignette detected – reloading real URL");
    await page.goto(href, { waitUntil: "domcontentloaded" });
  }
}

async function takeScreenshot(page: Page): Promise<string> {
  const baseDir = "D:/Kevin/Resume/Jobs/Cognizant/Client/VISA/Image";

  // Ensure directory exists
  if (!fs.existsSync(baseDir)) {
    fs.mkdirSync(baseDir, { recursive: true });
  }

  // Format date: ddMMyyyy_HHmmss
  const now = new Date();
  const dd = String(now.getDate()).padStart(2, "0");
  const MM = String(now.getMonth() + 1).padStart(2, "0");
  const yyyy = now.getFullYear();
  const HH = String(now.getHours()).padStart(2, "0");
  const mm = String(now.getMinutes()).padStart(2, "0");
  const ss = String(now.getSeconds()).padStart(2, "0");

  const fileName = `TypeScriptPlaywright_${dd}${MM}${yyyy}_${HH}${mm}${ss}.png`;
  const fullPath = path.join(baseDir, fileName);

  await page.screenshot({
    path: fullPath,
    fullPage: true,
  });

  console.log(`📸 Screenshot saved: ${fullPath}`);

  return fullPath;
}

async function uploadFile(
  page: Page,
  fileInputSelector: string,
  filePath: string
): Promise<void> {
  if (!fs.existsSync(filePath)) {
    throw new Error(`Upload failed. File does not exist: ${filePath}`);
  }

  await page.setInputFiles(fileInputSelector, filePath);

  console.log(`📤 Uploaded file: ${filePath}`);
}

async function downloadToFolder(
  page: Page,
  buttonSelector: string,
  saveDir: string
): Promise<string> {
  const [download] = await Promise.all([
    page.waitForEvent("download"),
    page.locator(buttonSelector).click(),
  ]);

  const fileName = download.suggestedFilename();
  const fullPath = path.join(saveDir, fileName);

  await download.saveAs(fullPath); // <-- THIS is what actually writes to disk
  console.log(`Downloaded saved to:\n${fullPath}`);

  return fullPath;
}

async function openImageAndClose(
  filePath: string,
  holdMs: number = 3000
): Promise<void> {
  // Open image with default Windows app
  exec(`start "" "${filePath}"`);

  // Wait X ms
  await new Promise((resolve) => setTimeout(resolve, holdMs));

  // Kill Windows Photos app
  exec(`taskkill /IM Photos.exe /F`);
}

async function dynamicProperties(page: Page) {
  await page
    .locator("//span[normalize-space()='Dynamic Properties']")
    .scrollIntoViewIfNeeded();
  await page.locator("//span[normalize-space()='Dynamic Properties']").click();
  await page.waitForURL(/dynamic-properties/);
  console.log("Opened Dynamic Properties page");

  // Restart timers
  await page.reload();
  await page.locator("#colorChange").waitFor({ state: "visible" });
  console.log("Refreshed Dynamic Properties page (timer restarted)");

  // Enable after 5 seconds
  const enableAfterBtn = page.locator("#enableAfter");
  console.log(
    "Enable button initially enabled? " + (await enableAfterBtn.isEnabled())
  );
  await enableAfterBtn.waitFor({ state: "visible" });
  await page.waitForTimeout(5500); // simple + stable for this demo
  console.log(
    "Enable button is now ENABLED? " + (await enableAfterBtn.isEnabled())
  );

  // Color change (class contains text-danger)
  const colorChangeBtn = page.locator("#colorChange");
  const classBefore = (await colorChangeBtn.getAttribute("class")) ?? "";
  console.log("Class BEFORE: " + classBefore);

  await page.waitForFunction(() => {
    const el = document.querySelector("#colorChange");
    return el && el.className.includes("text-danger");
  });

  const classAfter = (await colorChangeBtn.getAttribute("class")) ?? "";
  console.log("Class AFTER: " + classAfter);

  // Visible after 5 seconds
  console.log("Waiting for Visible After 5 Seconds button...");
  await page.locator("#visibleAfter").waitFor({ state: "visible" });
  console.log("Visible button is now DISPLAYED");

  console.log("Dynamic Properties demo completed");
}
