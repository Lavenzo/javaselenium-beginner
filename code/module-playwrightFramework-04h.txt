async function dismissDemoQAAds(page: Page): Promise<void> {
  // If page already closed, do nothing
  if (page.isClosed()) return;

  // 1) If vignette hash appears, try escape (no waiting)
  try {
    if (!page.isClosed() && page.url().includes("google_vignette")) {
      await page.keyboard.press("Escape");
    }
  } catch {}

  // If page got replaced/closed, stop
  if (page.isClosed()) return;

  // 2) Close fixed bottom banner if present (no waitForTimeout)
  try {
    const closeBtn = page.locator("#close-fixedban");
    if (!page.isClosed() && (await closeBtn.count()) > 0) {
      // Use first() to avoid strict-mode issues
      await closeBtn
        .first()
        .click({ timeout: 1000 })
        .catch(() => {});
    }
  } catch {}

  if (page.isClosed()) return;

  // 3) Hide common overlay containers (best-effort)
  try {
    await page.evaluate(() => {
      const selectors = [
        "#ad_position_box",
        'div[id^="google_ads_iframe_"]',
        'div[id$="__container__"]',
        'iframe[id^="google_ads_iframe_"]',
        'iframe[src*="googlesyndication"]',
      ];

      for (const sel of selectors) {
        document.querySelectorAll(sel).forEach((el) => {
          const node = el as HTMLElement;
          node.style.setProperty("display", "none", "important");
          node.style.setProperty("visibility", "hidden", "important");
          node.style.setProperty("pointer-events", "none", "important");
          node.style.setProperty("opacity", "0", "important");
        });
      }

      // restore scroll/click
      document.body.style.overflow = "auto";
      document.documentElement.style.overflow = "auto";
    });
  } catch {}
}

test("TC01 - Prerequisite Website Navigation to Menu Page", async ({
  page,
}, testInfo) => {
  testMetadata04h(testInfo);

  await page.goto("https://demoqa.com/menu", {
    waitUntil: "domcontentloaded",

  // dismiss Google ads if appear
  await dismissDemoQAAds(page);

  // 1️⃣ Page exists
  expect(page).toBeTruthy();

  // 2️⃣ Correct page loaded (auto-waits)
  await expect(page).toHaveURL(/\/menu/);

  // 3️⃣ Critical element exists (more stable)
  const heading = page.locator("h1.text-center");
  await expect(heading).toHaveText("Menu");
});

async function getHoveredMainItem1(page: Page): Promise<string> {
  try {
    // Locate "Main Item 1"
    const mainItem1 = page.locator("//a[text()='Main Item 1']");

    // Hover over Main Item 1
    await mainItem1.hover();

    // Small wait to allow hover UI to appear (if any)
    await page.waitForTimeout(1000);

    // Try to find any hovered / visible sub item
    const hoveredItems = page.locator("//ul[contains(@class,'submenu')]//a");
    const count = await hoveredItems.count();

    if (count === 0) return "";

    return (await hoveredItems.first().innerText()).trim();
  } catch {
    return "";
  }
}

async function getHoveredMainItem2(page: Page): Promise<string> {

    return "";
  } catch {
    return "";
  }
}

async function getHoveredSubSubList(page: Page): Promise<string> {
  try {
    // 1) Hover over Main Item 2
    const mainItem2 = page.locator("//a[text()='Main Item 2']");
    await mainItem2.hover();

    await page.waitForTimeout(800);

    // 2) Hover over SUB SUB LIST »
    // DemoQA: Main Item 2 submenu is a following-sibling <ul>
    const subSubList = page.locator(
      "//a[text()='Main Item 2']/following-sibling::ul/li/a[contains(text(),'SUB SUB LIST')]"
    );
    await subSubList.hover();

    await page.waitForTimeout(800);

    // 3) Check if there is a nested <ul> inside SUB SUB LIST »
    const nestedUl = page.locator(
      "//a[contains(text(),'SUB SUB LIST')]/parent::li/ul"
    );

    // 4) If there is, return the text of SUB SUB LIST »
    if ((await nestedUl.count()) > 0 && (await nestedUl.first().isVisible())) {
      const text = (await subSubList.innerText()).replace("»", "").trim();
      return text;
    }

    return "";
  } catch {
    return "";
  }
}

async function getHoveredSubSubItem1(page: Page): Promise<string> {
  try {
    // 1) Hover over Main Item 2
    const mainItem2 = page.locator("//a[normalize-space()='Main Item 2']");
    await mainItem2.scrollIntoViewIfNeeded();
    await mainItem2.hover();

    // Wait for Main Item 2 submenu UL (DemoQA: following-sibling::ul)
    const mainItem2Ul = page.locator(
      "//a[normalize-space()='Main Item 2']/following-sibling::ul"
    );
    await mainItem2Ul.first().waitFor({ state: "visible", timeout: 2000 });

    // 2) Hover over SUB SUB LIST »
    const subSubListLink = mainItem2Ul.locator(
      "xpath=.//a[contains(normalize-space(),'SUB SUB LIST')]"
    );
    await subSubListLink.hover();

    // 3) Wait for nested UL under SUB SUB LIST
    const subSubListLi = subSubListLink.locator("xpath=ancestor::li[1]");
    const nestedUl = subSubListLi.locator("xpath=.//ul");
    await nestedUl.first().waitFor({ state: "visible", timeout: 2000 });

    // 4) Hover over Sub Sub Item 1
    const subSubItem1 = nestedUl.locator(
      "xpath=.//a[normalize-space()='Sub Sub Item 1']"
    );
    await subSubItem1.first().waitFor({ state: "visible", timeout: 2000 });
    await subSubItem1.hover();

    // Return text
    return (await subSubItem1.innerText()).trim();
  } catch {
    return "";
  }
}
