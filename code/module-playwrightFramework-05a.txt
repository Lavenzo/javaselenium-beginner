async function dismissDemoQAAds(page: Page): Promise<void> {
  // If page already closed, do nothing
  if (page.isClosed()) return;

  // 1) If vignette hash appears, try escape (no waiting)
  try {
    if (!page.isClosed() && page.url().includes("google_vignette")) {
      await page.keyboard.press("Escape");
    }
  } catch {}

  // If page got replaced/closed, stop
  if (page.isClosed()) return;

  // 2) Close fixed bottom banner if present (no waitForTimeout)
  try {
    const closeBtn = page.locator("#close-fixedban");
    if (!page.isClosed() && (await closeBtn.count()) > 0) {
      // Use first() to avoid strict-mode issues
      await closeBtn
        .first()
        .click({ timeout: 1000 })
        .catch(() => {});
    }
  } catch {}

  if (page.isClosed()) return;

  // 3) Hide common overlay containers (best-effort)
  try {
    await page.evaluate(() => {
      const selectors = [
        "#ad_position_box",
        'div[id^="google_ads_iframe_"]',
        'div[id$="__container__"]',
        'iframe[id^="google_ads_iframe_"]',
        'iframe[src*="googlesyndication"]',
      ];

      for (const sel of selectors) {
        document.querySelectorAll(sel).forEach((el) => {
          const node = el as HTMLElement;
          node.style.setProperty("display", "none", "important");
          node.style.setProperty("visibility", "hidden", "important");
          node.style.setProperty("pointer-events", "none", "important");
          node.style.setProperty("opacity", "0", "important");
        });
      }

      // restore scroll/click
      document.body.style.overflow = "auto";
      document.documentElement.style.overflow = "auto";
    });
  } catch {}
}

const LIST_LABELS = ["One", "Two", "Three", "Four", "Five", "Six"] as const;
type ListLabel = (typeof LIST_LABELS)[number];

function getRandomPositions(): [ListLabel, ListLabel] {
  const idxOne = Math.floor(Math.random() * LIST_LABELS.length);
  let idxTwo = Math.floor(Math.random() * LIST_LABELS.length);
  while (idxTwo === idxOne)
    idxTwo = Math.floor(Math.random() * LIST_LABELS.length);
  const posOne = LIST_LABELS[idxOne]; // target
  const posTwo = LIST_LABELS[idxTwo]; // source
  return [posOne, posTwo];
}

function listItemByText(page: Page, label: string): Locator {
  // List tab pane items
  return page
    .locator("#demo-tabpane-list .list-group-item", { hasText: label })
    .first();
}

async function ensureListTabActive(page: Page) {
  const listTab = page.locator("#demo-tab-list");
  const isSelected = await listTab.getAttribute("aria-selected");
  if (isSelected !== "true") {
    await listTab.scrollIntoViewIfNeeded();
    await listTab.click();
    await expect(listTab).toHaveAttribute("aria-selected", "true");
  }
}

/**
 * Drag "sourceLabel" and drop above "targetLabel" in the List tab.
 * This uses mouse drag with deterministic drop point: top-half of target.
 */
async function moveElementListPosition(
  page: Page,
  sourceLabel: string,
  targetLabel: string,
) {
  const source = listItemByText(page, sourceLabel);
  const target = listItemByText(page, targetLabel);

  await expect(source).toBeVisible();
  await expect(target).toBeVisible();

  await source.scrollIntoViewIfNeeded();

  const srcBox = await source.boundingBox();
  const tgtBox = await target.boundingBox();

  if (!srcBox || !tgtBox) {
    throw new Error(
      `Bounding box missing. source=${sourceLabel}, target=${targetLabel}`,
    );
  }

  // Start from center of source
  const startX = srcBox.x + srcBox.width / 2;
  const startY = srcBox.y + srcBox.height / 2;

  // Drop slightly above the center of target (top-half), to land ABOVE it
  const dropX = tgtBox.x + tgtBox.width / 2;
  const dropY = tgtBox.y + Math.max(3, tgtBox.height * 0.25);

  await page.mouse.move(startX, startY);
  await page.mouse.down();
  await page.waitForTimeout(200);

  // Move to target
  await page.mouse.move(dropX, dropY, { steps: 12 });
  await page.waitForTimeout(300);

  await page.mouse.up();
}

/** Returns current visual order of list items (text array) */
async function getListVisualOrder(page: Page): Promise<string[]> {
  const items = page.locator("#demo-tabpane-list .list-group-item");
  const count = await items.count();
  const order: string[] = [];

  for (let i = 0; i < count; i++) {
    order.push((await items.nth(i).innerText()).trim());
  }
  return order;
}

  page,
}, testInfo) => {
  testMetadata05a_pt1(testInfo);
  console.log("===== START TC02 : Random Element Reordering =====");

  // If your suite already navigates in a prereq, you can remove this line
  await page.goto("https://demoqa.com/sortable");

  // Ensure List tab active
  await ensureListTabActive(page);
  console.log("List tab is active");

  // 1) Randomize two positions
  const [posOne, posTwo] = getRandomPositions(); // posOne = target, posTwo = source
  console.log(`Randomised source element (PosTwo): ${posTwo}`);
  console.log(`Randomised target element (PosOne): ${posOne}`);
  console.log(`Action: Move [${posTwo}] above [${posOne}]`);

  let movedCorrectly = false;

  for (let attempt = 1; attempt <= 3; attempt++) {
    console.log(`Attempt ${attempt} - Drag [${posTwo}] onto [${posOne}]`);

    await page.waitForTimeout(800);
    console.log("Waiting for UI to stabilise after drag-and-drop");

    const order = await getListVisualOrder(page);

    let posTwoIndex = order.indexOf(posTwo);
    let posOneIndex = order.indexOf(posOne);

    order.forEach((txt, i) => console.log(`Current position [${i}] : ${txt}`));

    console.log(
      `Index check (Attempt ${attempt}) -> ${posTwo}: ${posTwoIndex} | ${posOne}: ${posOneIndex}`,
    );

    if (posTwoIndex !== -1 && posOneIndex !== -1 && posTwoIndex < posOneIndex) {
      movedCorrectly = true;
      console.log(`Move successful on attempt ${attempt}`);
      break;
    } else {
      console.log("Move not in expected order yet, retrying...");
    }
  }

  expect(
    movedCorrectly,
    `Expected ${posTwo} to be above ${posOne}`,
  ).toBeTruthy();

  console.log(`Verification PASSED: [${posTwo}] is above [${posOne}]`);
  console.log("===== END TC02 =====");
});

  if (page.isClosed()) return;

  // 2) Close fixed bottom banner if present (no waitForTimeout)
  try {
    const closeBtn = page.locator("#close-fixedban");
    if (!page.isClosed() && (await closeBtn.count()) > 0) {
      // Use first() to avoid strict-mode issues
      await closeBtn
        .first()
        .click({ timeout: 1000 })
        .catch(() => {});
    }
  } catch {}

  if (page.isClosed()) return;

  // 3) Hide common overlay containers (best-effort)
  try {
    await page.evaluate(() => {
      const selectors = [
        "#ad_position_box",
        'div[id^="google_ads_iframe_"]',
        'div[id$="__container__"]',
        'iframe[id^="google_ads_iframe_"]',
        'iframe[src*="googlesyndication"]',
      ];

      for (const sel of selectors) {
        document.querySelectorAll(sel).forEach((el) => {
          const node = el as HTMLElement;
          node.style.setProperty("display", "none", "important");
          node.style.setProperty("visibility", "hidden", "important");
          node.style.setProperty("pointer-events", "none", "important");
          node.style.setProperty("opacity", "0", "important");
        });
      }

      // restore scroll/click
      document.body.style.overflow = "auto";
      document.documentElement.style.overflow = "auto";
    });
  } catch {}
}

/** Get Grid visual order (reads DOM order of grid items, like your Selenium method) */
async function getGridVisualOrder(page: Page): Promise<string[]> {
  await ensureOnSortablePage(page);
  const items = page.locator(
    "#demo-tabpane-grid .create-grid .list-group-item",
  );
  const texts = await items.allTextContents();
  return texts.map((t) => t.trim()).filter(Boolean);
}

/** Random pick two distinct labels from 1..9 */
function getRandomTwoGridLabels(): [string, string] {
  const labels = [
    "One",
    "Two",
    "Three",
    "Four",
    "Five",
    "Six",
    "Seven",
    "Eight",
    "Nine",
  ];
  const idx1 = Math.floor(Math.random() * labels.length);
  let idx2 = idx1;
  while (idx2 === idx1) idx2 = Math.floor(Math.random() * labels.length);
  return [labels[idx1], labels[idx2]];
}
/**
 * TEXT SWAP workaround (DOM text manipulation) - matches your final working approach:
 * swap a.textContent and b.textContent, return true/false.
 */
export async function swapGridItemsByText(
  page: Page,
  aText: string,
  bText: string,
): Promise<boolean> {
  await ensureOnSortablePage(page);

  const swapped = await page.evaluate(
    ([a, b]) => {
      const grid = document.querySelector("#demo-tabpane-grid .create-grid");
      if (!grid) return false;

      const items = Array.from(grid.querySelectorAll(".list-group-item"));
      const elA = items.find((el) => (el.textContent || "").trim() === a);
      const elB = items.find((el) => (el.textContent || "").trim() === b);

      if (!elA || !elB) return false;

      const temp = elA.textContent;
      elA.textContent = elB.textContent;
      elB.textContent = temp;

      return true;
    },
    [aText, bText] as const,
  );

  return Boolean(swapped);
}

export async function ensureGridTabActive(page: Page): Promise<void> {
  await ensureOnSortablePage(page);

  const listTab = page.locator("#demo-tab-list");
  const gridTab = page.locator("#demo-tab-grid");

  // Force state change: click List first, then Grid
  await listTab.click({ timeout: 10000 }).catch(async () => {
    await page
      .locator("#demo-tab-list")
      .click({ force: true })
      .catch(() => {});
  });

  await expect(listTab).toHaveAttribute("aria-selected", "true", {
    timeout: 15000,
  });

  await gridTab.click({ timeout: 10000 }).catch(async () => {
    await page
      .locator("#demo-tab-grid")
      .click({ force: true })
      .catch(() => {});
  });

  // This sometimes needs longer because DemoQA tab state can be sluggish
  await expect(gridTab).toHaveAttribute("aria-selected", "true", {
    timeout: 20000,
  });

  // Bonus: ensure grid container exists (prevents “tabs selected but grid not ready”)
  await expect(page.locator("#demo-tabpane-grid .create-grid")).toBeVisible({
    timeout: 15000,
  });
}

async function ensureOnSortablePage(page: Page): Promise<void> {
  if (page.isClosed()) return;

  const url = page.url();

  // Hard-navigate if not on Sortable OR caught in vignette
  if (!url.includes("demoqa.com/sortable") || url.includes("google_vignette")) {
    await page.goto("https://demoqa.com/sortable", {
      waitUntil: "domcontentloaded",
    });
  }

  await dismissDemoQAAds(page);

  // Tabs must exist (key flake guard)
  await expect(page.locator("#demo-tab-list")).toBeVisible({ timeout: 15000 });
  await expect(page.locator("#demo-tab-grid")).toBeVisible({ timeout: 15000 });
}
