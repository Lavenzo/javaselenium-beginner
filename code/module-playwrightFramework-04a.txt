// Test File
import { test, expect, type Page } from "@playwright/test";
import { testMetadata04a } from "./utils/testMetadata04";

test.describe("TC01 - Prerequisite Website Navigation", () => {
  test("verifyPrereqIsWorking", async ({ page }, testInfo) => {
    testMetadata04a(testInfo);

    await page.goto("https://demoqa.com/accordian", {
      waitUntil: "domcontentloaded",
    });

    // 1️⃣ Page exists
    expect(page).toBeTruthy();

    // 2️⃣ Correct page loaded (auto-waits)
    await expect(page).toHaveURL(/\/accordian/);

    // 3️⃣ Critical element exists (more stable)
    const heading = page.locator("h1.text-center");
    await expect(heading).toHaveText("Accordian");
  });
});

async function getFirstAccordian(
  page: import("@playwright/test").Page
): Promise<string> {
  const firstAccordion = page.locator("#section1Heading");

  await firstAccordion.scrollIntoViewIfNeeded();

  try {
    await firstAccordion.click();
  } catch {
    await firstAccordion.click({ force: true });
  }

  const accordionText = page.locator("#section1Content p");
  await expect(accordionText).toBeVisible();

  return (await accordionText.textContent())?.trim() ?? "";
}

test.describe("TC02 - Verify First Accordion", () => {
  test("verifyFirstAccordian", async ({ page }, testInfo) => {
    testMetadata04a(testInfo);

    await page.goto("https://demoqa.com/accordian", {
      waitUntil: "domcontentloaded",
    });

    const actualText = await getFirstAccordian(page);

    const expectedText =
      "Lorem Ipsum is simply dummy text of the printing and typesetting industry. " +
      "Lorem Ipsum has been the industry's standard dummy text ever since the 1500s, when an unknown printer took a " +
      "galley of type and scrambled it to make a type specimen book. It has survived not only five centuries, but also " +
      "the leap into electronic typesetting, remaining essentially unchanged. It was popularised in the 1960s with the release of " +
      "Letraset sheets containing Lorem Ipsum passages, and more recently with desktop publishing software like Aldus PageMaker " +
      "including versions of Lorem Ipsum.";

    await expect(actualText, { message: "First Accordian text mismatch" }).toBe(
      expectedText
    );
  });
});

// --- Helper: normalize (same idea as your Selenium normalizeText) ---
function normalizeText(s: string): string {
  return s
    .replace(/\u00A0/g, " ") // NBSP -> normal space
    .replace(/\s+/g, " ") // collapse whitespace/newlines
    .trim();
}

// --- Helper: getSecondAccordian (actions + return text, no assertion inside) ---
async function getSecondAccordian(
  page: import("@playwright/test").Page
): Promise<string> {
  // 1) Click second accordion header
  const secondAccordian = page.locator("#section2Heading");

  await secondAccordian.scrollIntoViewIfNeeded();

  try {
    await secondAccordian.click();
  } catch {
    await secondAccordian.click({ force: true });
  }

  // 2) Get ALL paragraph texts inside second accordion
  const paragraphs = page.locator("#section2Content p");
  await expect(paragraphs.first()).toBeVisible();

  const texts = await paragraphs.allTextContents();

  // 3) Combine into a single string (same as StringBuilder append + space)
  return texts
    .map((t) => t.trim())
    .join(" ")
    .trim();
}

// --- Test: verifySecondAccordian ---
test.describe("TC03 - Verify Second Accordion", () => {
  test("verifySecondAccordian", async ({ page }, testInfo) => {
    testMetadata04a(testInfo);

      "at Hampden-Sydney College in Virginia, looked up one of the more obscure Latin words, consectetur, from a Lorem Ipsum " +
      "passage, and going through the cites of the word in classical literature, discovered the undoubtable source. Lorem Ipsum " +
      'comes from sections 1.10.32 and 1.10.33 of "de Finibus Bonorum et Malorum" (The Extremes of Good and Evil) by Cicero, ' +
      "written in 45 BC. This book is a treatise on the theory of ethics, very popular during the Renaissance. " +
      'The first line of Lorem Ipsum, "Lorem ipsum dolor sit amet..", comes from a line in section 1.10.32. The standard chunk ' +

      "versions from the 1914 translation by H. Rackham.";

    await expect(normalizeText(actual), {

    }).toBe(normalizeText(expected));
  });
});

): Promise<string> {

  const thirdAccordian = page.locator("#section3Heading");

  await thirdAccordian.scrollIntoViewIfNeeded();

  try {
    await thirdAccordian.click();
  } catch {
    await thirdAccordian.click({ force: true });
  }

  // 2️⃣ Wait for accordion content text (partial text match like Selenium contains())
  const accordionText = page.locator("#section3Content p", {
    hasText: "It is a long established fact that a reader will",
  });

  await expect(accordionText).toBeVisible();

  // 3️⃣ Return text
  return (await accordionText.textContent())?.trim() ?? "";
}
