async function stopProgressBarInRange(
  page: Page,
  minPercent: number,
  maxPercent: number
): Promise<number> {
  // Navigate (keep here or do it in your test – your choice)
  await page.goto("https://demoqa.com/progress-bar", {
    waitUntil: "domcontentloaded",
  });

  const startStopBtn = page.locator("#startStopButton");
  await expect(startStopBtn).toBeVisible();

  // Start progress
  await startStopBtn.click();

  const progressBar = page.locator("div[role='progressbar']");
  await expect(progressBar).toBeVisible();

  const timeoutMs = 15000; // safety timeout
  const startTime = Date.now();

  let currentValue = -1;

  while (Date.now() - startTime < timeoutMs) {
    // aria-valuenow is the live percentage
    const value = await progressBar.getAttribute("aria-valuenow");

    if (value && value.trim() !== "") {
      const parsed = Number(value.trim());
      if (!Number.isNaN(parsed)) {
        currentValue = parsed;
        console.log(`Current progress: ${currentValue}%`);

        // Stop immediately when within range
        if (currentValue >= minPercent && currentValue <= maxPercent) {
          await startStopBtn.click();
          console.log(`Stopped at: ${currentValue}%`);
          return currentValue;
        }
      }
    }

    // small poll interval (similar to your 80ms sleep)
    await page.waitForTimeout(80);
  }

  // If we reach here, we didn't hit range in time - try stop anyway, then fail
  try {
    await startStopBtn.click();
  } catch {
    // ignore
  }

  throw new Error(
    `Did not hit range ${minPercent}-${maxPercent}% within timeout.`
  );
}

test("TC02 - Verify Positive Test Scenario with 20 Percent Range Checks", async ({
  page,
}, testInfo) => {
  // 1) Click Start (done inside helper)
  // 2) Poll until value is within 20-30
  // 3) Click Stop immediately when within range
  // 4) Assert final stopped value is within range + print exact result
  testMetadata04e(testInfo);
  const stoppedAt = await stopProgressBarInRange(page, 20, 30);

  console.log(`Final stopped progress (for assertion): ${stoppedAt}%`);

  expect(
    stoppedAt >= 20 && stoppedAt <= 30,
    `Progress bar stopped at ${stoppedAt}% (expected between 20% and 30%)`
  ).toBeTruthy();
});

test("TC03 - Verify Positive Test Scenario with 70 Percent Range Checks", async ({

  // 1) Click Start (done inside helper)

async function runProgressBarUntilComplete(page: Page): Promise<void> {
  const startStopBtn = page.locator("#startStopButton");
  const progressBar = page.locator("div[role='progressbar']");
  const progressText = page.locator("#progressBar");

  // Start
  await startStopBtn.click();

  // Wait until aria-valuenow reaches 100 (most reliable)
  await expect
    .poll(
      async () => {
        const val = await progressBar.getAttribute("aria-valuenow");
        return val ? parseInt(val.trim(), 10) : -1;
      },
      {
        timeout: 30000, // allow enough time to reach 100
        message: "Progress bar did not reach 100% within timeout",
      }
    )
    .toBe(100);

  // Optional: confirm UI text also shows 100%
  await expect(progressText).toContainText("100%", { timeout: 5000 });
}

test("TC04 - Verify Negative Test Scenario for Button Label After Completion", async ({
  page,
}, testInfo) => {
  await page.goto("https://demoqa.com/progress-bar", {
    waitUntil: "domcontentloaded",
  });

  testMetadata04e(testInfo);

  // 1) Run until complete
  await runProgressBarUntilComplete(page);

  // 2) Reset button should now be visible
  const resetButton = page.locator("#resetButton");
  await expect(resetButton).toBeVisible({ timeout: 10000 });

  // 3) Get button label
  const actualLabel = (await resetButton.textContent())?.trim() ?? "";
  const expectedLabel = "Reset Progress";

  console.log(`Button label after completion: ${actualLabel}`);

  // 4) Assert (INTENTIONAL FAILURE)
  expect(actualLabel, "Button label mismatch after progress reaches 100%").toBe(
    expectedLabel
  );
});
