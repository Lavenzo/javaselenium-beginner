import { test, expect, type Page, Locator } from "@playwright/test";
import { testMetadata05b_pt1 } from "./utils/testMetadata05b";

async function dismissDemoQAAds(page: Page): Promise<void> {
  // If page already closed, do nothing
  if (page.isClosed()) return;

  // 1) If vignette hash appears, try escape (no waiting)
  try {
    if (!page.isClosed() && page.url().includes("google_vignette")) {
      await page.keyboard.press("Escape");
    }
  } catch {}

  // If page got replaced/closed, stop
  if (page.isClosed()) return;

  // 2) Close fixed bottom banner if present (no waitForTimeout)
  try {
    const closeBtn = page.locator("#close-fixedban");
    if (!page.isClosed() && (await closeBtn.count()) > 0) {
      // Use first() to avoid strict-mode issues
      await closeBtn
        .first()
        .click({ timeout: 1000 })
        .catch(() => {});
    }
  } catch {}

  if (page.isClosed()) return;

  // 3) Hide common overlay containers (best-effort)
  try {
    await page.evaluate(() => {
      const selectors = [
        "#ad_position_box",
        'div[id^="google_ads_iframe_"]',
        'div[id$="__container__"]',
        'iframe[id^="google_ads_iframe_"]',
        'iframe[src*="googlesyndication"]',
      ];

      for (const sel of selectors) {
        document.querySelectorAll(sel).forEach((el) => {
          const node = el as HTMLElement;
          node.style.setProperty("display", "none", "important");
          node.style.setProperty("visibility", "hidden", "important");
          node.style.setProperty("pointer-events", "none", "important");
          node.style.setProperty("opacity", "0", "important");
        });
      }

      // restore scroll/click
      document.body.style.overflow = "auto";
      document.documentElement.style.overflow = "auto";
    });
  } catch {}
}

test("TC01 - Prerequisite Website Navigation to Selectable Page - List View", async ({

}, testInfo) => {
  testMetadata05b_pt1(testInfo);

  await page.goto("https://demoqa.com/selectable", {

});

// ---------- helpers (reusable) ----------
function pickRandom<T>(arr: T[]): T {
  return arr[Math.floor(Math.random() * arr.length)];
}

const byXpath = (page: Page, xp: string) => page.locator(`xpath=${xp}`);

async function clickTabWithFallback(page: Page, tabId: string, tcId: string) {
  const tab = page.locator(`#${tabId}`);
  await expect(tab, `${tcId} - Tab not found: #${tabId}`).toHaveCount(1);

  // normal click first
  await tab.click().catch(() => {});
  const ariaAfter = (await tab.getAttribute("aria-selected")) ?? "false";
  console.log(
    `${tcId} - ${tabId} aria-selected after normal click: ${ariaAfter}`,
  );

  // JS click fallback if still not selected
  if (ariaAfter !== "true") {
    console.log(`${tcId} - Clicking ${tabId} using JS fallback`);
    await page.evaluate((id) => {
      const el = document.getElementById(id) as HTMLElement | null;
      el?.click();
    }, tabId);
  }

  // final wait until selected
  await expect(
    tab,
    `${tcId} - ${tabId} did not become selected`,
  ).toHaveAttribute("aria-selected", "true");
}

async function ensureListTabReady(page: Page, tcId: string) {
  // Navigate (IMPORTANT)
  await page.goto("https://demoqa.com/selectable", {
    waitUntil: "domcontentloaded",
  });
  await expect(page, `${tcId} - Not on /selectable`).toHaveURL(/\/selectable/);

  // Ensure List tab is active
  await clickTabWithFallback(page, "demo-tab-list", tcId);

  // Ensure list panel is actually loaded
  const container = page.locator("#verticalListContainer");
  await expect(container, `${tcId} - List container not visible`).toBeVisible();
  await expect(
    container.locator("li"),
    `${tcId} - List items not loaded`,
  ).toHaveCount(4);

  // Optional: scroll container into view (helps if ads/viewport)
  await container.scrollIntoViewIfNeeded();
}

async function clearAllActiveSelections(
  page: Page,
  xpaths: string[],
  tcId: string,
) {
  await test.step(`${tcId} - Clear any active selections (baseline reset)`, async () => {
    for (const xp of xpaths) {
      const loc = byXpath(page, xp);

      // Existence check (more accurate than visibility during resets)
      await expect(
        loc,
        `${tcId} - Element not found during reset: ${xp}`,
      ).toHaveCount(1);

      await loc.scrollIntoViewIfNeeded();

      const cls = (await loc.getAttribute("class")) ?? "";
      const isActive = cls.includes("active");
      console.log(
        `${tcId} - Reset check | active=${isActive} | class=${cls} | XPath=${xp}`,
      );

      if (isActive) {
        console.log(`${tcId} - Unselecting active item: ${xp}`);
        await loc.click();
        await expect(loc).not.toHaveClass(/active/);
      }
    }
  });
}

async function ensureSelected(page: Page, xp: string, tcId: string) {
  const loc = byXpath(page, xp);

  await test.step(`${tcId} - Ensure item is selected`, async () => {
    await expect(loc, `${tcId} - Element not found: ${xp}`).toHaveCount(1);
    await loc.scrollIntoViewIfNeeded();

    const beforeCls = (await loc.getAttribute("class")) ?? "";
    console.log(`${tcId} - Class BEFORE click: ${beforeCls} | XPath: ${xp}`);

    // If active already, unselect first (so your test is deterministic)
    if (beforeCls.includes("active")) {
      console.log(
        `${tcId} - Already active, clicking to unselect first: ${xp}`,
      );
      await loc.click();
      await expect(loc).not.toHaveClass(/active/);
    }

    console.log(`${tcId} - Clicking to select: ${xp}`);
    await loc.click();
    await expect(loc, `${tcId} - Expected item to become active`).toHaveClass(
      /active/,
    );
  });
}
