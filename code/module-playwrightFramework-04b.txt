// Test File
import { test, expect, type Page } from "@playwright/test";
import { testMetadata04b } from "./utils/testMetadata04";

test.describe("TC01 - Prerequisite Website Navigation", () => {
  test("verifyPrereqIsWorking", async ({ page }, testInfo) => {
    testMetadata04b(testInfo);

    await page.goto("https://demoqa.com/auto-complete", {
      waitUntil: "domcontentloaded",
    });

    // 1️⃣ Page exists
    expect(page).toBeTruthy();

    // 2️⃣ Correct page loaded (auto-waits)
    await expect(page).toHaveURL(/\/auto-complete/);

    // 3️⃣ Critical element exists (more stable)
    const heading = page.locator("h1.text-center");
    await expect(heading).toHaveText("Auto Complete");
  });
});

async function get2MultiColour(page: Page): Promise<string> {
  // 1. Locate the multi auto-complete input
  const input = page.locator("#autoCompleteMultipleInput");

  // 2. Scroll into view (similar to JavascriptExecutor)
  await input.scrollIntoViewIfNeeded();

  // 3. Click + enter two colours
  await input.click();
  await input.fill("Yellow");
  await input.press("Enter");

  await input.fill("Blue");
  await input.press("Enter");

  // 4. Wait until both chips are visible
  const chips = page.locator("div.auto-complete__multi-value__label");
  await expect(chips).toHaveCount(2);

  // 5. Collect selected chip labels
  const selected = await chips.allTextContents();

  // 6. Return as a single string (same behavior as Java)
  return selected.map((text) => text.trim()).join(", ");
}

test("TC02 - Verify Multi Colour Names with 2 colours selected", async ({
  page,
}, testInfo) => {
  testMetadata04b(testInfo);
  await page.goto("https://demoqa.com/auto-complete");

  const selectedText = await get2MultiColour(page);

  expect(selectedText).toContain("Yellow");
  expect(selectedText).toContain("Blue");
});

async function get3MultiColour(page: Page): Promise<string> {
  const input = page.locator("#autoCompleteMultipleInput");

  await input.scrollIntoViewIfNeeded();
  await input.click();

  // Yellow
  await input.type("Yellow", { delay: 50 });
  await input.press("ArrowDown");
  await input.press("Enter");

  // Blue
  await input.type("Blue", { delay: 50 });
  await input.press("ArrowDown");
  await input.press("Enter");

  // Voilet (DemoQA spelling)
  await input.type("Voilet", { delay: 50 });
  await input.press("ArrowDown");
  await input.press("Enter");

  // Wait until all chips are rendered
  const chips = page.locator("div.auto-complete__multi-value__label");
  await expect(chips).toHaveCount(3);

  const selected = await chips.allTextContents();
  return selected.map((t) => t.trim()).join(", ");
}

test("TC03 - Verify Multi Colour Names with 3 colours selected", async ({
  page,
}, testInfo) => {
  testMetadata04b(testInfo);
  await page.goto("https://demoqa.com/auto-complete");

  const selectedText = await get3MultiColour(page);

  expect(selectedText).toContain("Yellow");
  expect(selectedText).toContain("Blue");
  expect(selectedText).toContain("Orange"); // intentionally wrong
});

async function getSingleColour(page: Page): Promise<string> {
  // Navigate only if not already on the page
  if (!page.url().includes("/auto-complete")) {
    await page.goto("https://demoqa.com/auto-complete");
  }

  // Single input box
  const input = page.locator("#autoCompleteSingleInput");

  // Scroll + click (Playwright-safe)
  await input.scrollIntoViewIfNeeded();
  await input.click();

  // Clear existing value (Ctrl + A → Backspace)
  await input.press("Control+A");
  await input.press("Backspace");

  // Type colour
  await input.type("Indigo", { delay: 50 });

  // Wait for dropdown option and click
  const indigoOption = page.locator("div.auto-complete__option", {
    hasText: "Indigo",
  });

  await expect(indigoOption).toBeVisible();
  await indigoOption.click();

  // Read rendered selected value (React single value)
  const singleValue = page.locator("div.auto-complete__single-value");
  await expect(singleValue).toBeVisible();

  return (await singleValue.textContent())!.trim();
}

test("TC04 - Verify Single Colour Name with 1 colour selected", async ({
  page,
}, testInfo) => {
  testMetadata04b(testInfo);

  const selectedText = await getSingleColour(page);

  console.log("Successfully managed to get Single Colour:", selectedText);
  expect(selectedText).toBe("Indigo");
});
